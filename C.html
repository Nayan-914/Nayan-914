			c=c+1;

3. Check if a Number is Palindrome

#include<stdio.h>
int main()
{
    int n, original, reverse = 0;
    printf("Enter a number: ");
    scanf("%d", &n);
    original = n;
    
    while(n > 0) {
        reverse = reverse * 10 + n % 10;
        n /= 10;
    }
    
    if(original == reverse)
        printf("%d is palindrome", original);
    else
        printf("%d is not palindrome", original);
    return 0;
}

Output for Palindrome Check

5. Check if a Number is Perfect

#include<stdio.h>
int main()
{
    int n, sum = 0;
    printf("Enter a number: ");
    scanf("%d", &n);
    
    for(int i = 1; i < n; i++) {
        if(n % i == 0)
            sum += i;
    }
    
    if(sum == n)
        printf("%d is perfect", n);
    else
        printf("%d is not perfect", n);
    return 0;
}

Output for Perfect Number Check

6. Sum of Both Diagonals in 3x3 Matrix

#include <stdio.h>
int main() {
    int a[3][3], sum = 0, i, j;

    printf("Enter 9 elements:\n");
    for(i = 0; i < 3; i++) {
        for(j = 0; j < 3; j++) {
            printf("Enter element [%d][%d]: ", i, j);
            scanf("%d", &a[i][j]);
        }
    }

    for(i = 0; i < 3; i++) {
        for(j = 0; j < 3; j++) {
            if(i == j || i + j == 2) {
                sum = sum + a[i][j];
            }
        }
    }

    printf("Sum of both diagonals: %d\n", sum);
    return 0;
}

Output for Diagonal Sum

7. Sum of Odd and Even Numbers in 3x3 Matrix

#include <stdio.h>
int main() {
    int a[3][3], odd = 0, even = 0, i, j;

    printf("Enter 9 elements:\n");
    for(i = 0; i < 3; i++) {
        for(j = 0; j < 3; j++) {
            printf("Enter element [%d][%d]: ", i, j);
            scanf("%d", &a[i][j]);
        }
    }

    for(i = 0; i < 3; i++) {
        for(j = 0; j < 3; j++) {
            if(a[i][j] % 2 == 0) {
                even = even + a[i][j];
            } else {
                odd = odd + a[i][j];
            }
        }
    }

    printf("Odd Sum: %d\n", odd);
    printf("Even Sum: %d\n", even);
    return 0;
}

Output for Odd/Even Sum
